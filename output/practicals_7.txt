// ======================================================
// 1. ABSTRACT CONTRACT
// ======================================================

abstract contract Animal {

    string public name;

    constructor(string memory _name) {
        name = _name;
    }

    /*
        This function has no implementation.

        The child contract MUST implement it.
    */
    function makeSound()
        public
        virtual
        returns (string memory);
}


// ======================================================
// 2. CHILD CONTRACT IMPLEMENTING ABSTRACT FUNCTION
// ======================================================

contract Dog is Animal {

    constructor(string memory _name)
        Animal(_name)
    {}

    function makeSound()
        public
        pure
        override
        returns (string memory)
    {
        return "Woof!";
    }
}


// ======================================================
// 3. ANOTHER CHILD CONTRACT
// ======================================================

contract Cat is Animal {

    constructor(string memory _name)
        Animal(_name)
    {}

    function makeSound()
        public
        pure
        override
        returns (string memory)
    {
        return "Meow!";
    }
}


// ======================================================
// 4. ABSTRACT PAYMENT CONTRACT
// ======================================================

abstract contract Payment {

    /*
        Common payment functionality
        can be placed in the parent.
    */

    function pay()
        public
        payable
        virtual;

    function paymentMessage()
        public
        pure
        returns (string memory)
    {
        return "Payment system";
    }
}


// ======================================================
// 5. IMPLEMENTING ABSTRACT PAYMENT CONTRACT
// ======================================================

contract MyPayment is Payment {

    uint public totalReceived;

    function pay()
        public
        payable
        override
    {
        totalReceived += msg.value;
    }
}


// ======================================================
// 6. INTERFACE
// ======================================================

interface ICalculator {

    function add(
        uint a,
        uint b
    )
        external
        pure
        returns (uint);

    function subtract(
        uint a,
        uint b
    )
        external
        pure
        returns (uint);
}


// ======================================================
// 7. CONTRACT IMPLEMENTING INTERFACE
// ======================================================

contract Calculator is ICalculator {

    function add(
        uint a,
        uint b
    )
        external
        pure
        override
        returns (uint)
    {
        return a + b;
    }

    function subtract(
        uint a,
        uint b
    )
        external
        pure
        override
        returns (uint)
    {
        return a - b;
    }
}


// ======================================================
// 8. CALLING ANOTHER CONTRACT
// ======================================================

contract CalculatorCaller {

    /*
        We don't need the complete Calculator contract.

        We only need to know the functions we want to call.
    */

    ICalculator public calculator;

    constructor(address _calculator) {
        calculator = ICalculator(_calculator);
    }

    function calculateAdd(
        uint a,
        uint b
    )
        public
        view
        returns (uint)
    {
        return calculator.add(a, b);
    }

    function calculateSubtract(
        uint a,
        uint b
    )
        public
        view
        returns (uint)
    {
        return calculator.subtract(a, b);
    }
}


// ======================================================
// 9. CONTRACT-TO-CONTRACT STATE READING
// ======================================================

contract Bank {

    mapping(address => uint) public balances;

    function deposit()
        public
        payable
    {
        balances[msg.sender] += msg.value;
    }

    function getBalance(address user)
        public
        view
        returns (uint)
    {
        return balances[user];
    }
}


// ======================================================
// 10. INTERFACE FOR BANK
// ======================================================

interface IBank {

    function getBalance(address user)
        external
        view
        returns (uint);

    function deposit()
        external
        payable;
}


// ======================================================
// 11. CONTRACT READING ANOTHER CONTRACT
// ======================================================

contract BankReader {

    IBank public bank;

    constructor(address _bank) {
        bank = IBank(_bank);
    }

    function checkBalance(address user)
        public
        view
        returns (uint)
    {
        return bank.getBalance(user);
    }
}


// ======================================================
// 12. CONTRACT CALLING ANOTHER CONTRACT TO CHANGE STATE
// ======================================================

contract CounterContract {

    uint public count;

    function increment()
        public
    {
        count++;
    }

    function decrement()
        public
    {
        count--;
    }
}


// ======================================================
// 13. INTERFACE FOR COUNTER
// ======================================================

interface ICounter {

    function increment()
        external;

    function decrement()
        external;

    function count()
        external
        view
        returns (uint);
}


// ======================================================
// 14. COUNTER CALLER
// ======================================================

contract CounterCaller {

    ICounter public counter;

    constructor(address _counter) {
        counter = ICounter(_counter);
    }

    function increaseCounter()
        public
    {
        counter.increment();
    }

    function decreaseCounter()
        public
    {
        counter.decrement();
    }

    function readCounter()
        public
        view
        returns (uint)
    {
        return counter.count();
    }
}


// ======================================================
// 15. MULTI-CONTRACT ARCHITECTURE
// ======================================================

contract TokenStorage {

    mapping(address => uint) public balances;

    function setBalance(
        address user,
        uint amount
    )
        public
    {
        balances[user] = amount;
    }

    function getBalance(
        address user
    )
        public
        view
        returns (uint)
    {
        return balances[user];
    }
}
