pragma solidity ^0.8.0;

contract FunctionTypesDemo {
    uint private privateVar;
    uint internal internalVar;
    uint public publicVar;

    function publicFunction() public {
        privateVar = 1;
        internalVar = 2;
        publicVar = 3;
    }

    function privateFunction() private {
        privateVar = 10;
    }

    function internalFunction() internal {
        internalVar = 20;
    }

    function externalFunction() external {
        publicVar = 30;
    }

    function callInternal() public {
        internalFunction();
    }

    function callPrivate() public {
        privateFunction();
    }
}

contract RequireDemo {
    uint public balance;

    function withdraw(uint _amount) public {
        require(_amount > 0, "Amount must be positive");
        require(_amount <= balance, "Insufficient balance");
        balance -= _amount;
    }

    function setBalance(uint _newBalance) public {
        require(_newBalance >= 0, "Balance cannot be negative");
        balance = _newBalance;
    }
}

contract RevertDemo {
    uint public balance;

    function withdraw(uint _amount) public {
        if (_amount <= 0) {
            revert("Amount must be positive");
        }
        if (_amount > balance) {
            revert("Insufficient balance");
        }
        balance -= _amount;
    }

    function setBalance(uint _newBalance) public {
        if (_newBalance < 0) {
            revert("Balance cannot be negative");
        }
        balance = _newBalance;
    }
}
