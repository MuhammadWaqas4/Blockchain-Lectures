// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;


library MathLibrary3 {

   function square(uint x)
       internal
       pure
       returns(uint)
   {
       return x * x;
   }
}

contract Calculator_lib {

    using MathLibrary3 for uint;

   function calculateSquare(uint x)
       public
       pure
       returns(uint)
   {
       return x.square();
   }
}
