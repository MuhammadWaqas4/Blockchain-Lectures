contract Bank {

    IERC20 public token; // IERC20

    // User => deposited token balance
    mapping(address => uint256) public balances;

    // Total fees collected by the vault
    uint256 public collectedFees;

    address public  admin = 0xAb8483F64d9C6d1EcF9b849Ae677dD3315835cb2;

    // 200 basis points = 2%
    uint256 public constant FEE_BPS = 200; // 2

    // 10000 basis points = 100%
    uint256 public constant BPS = 10000; // 100

    constructor(address _token) {
        token = IERC20(_token);        
    }

    modifier onlyAdmin() {
        require(msg.sender == admin, "Not admin");
        _;
    }
        
    function changeAdmin(address newAdmin) external onlyAdmin {
        require(newAdmin != address(0), "Invalid admin");
        admin = newAdmin;
    }

    // ==================================================
    // 1. NORMAL DEPOSIT — NO FEE
    // ==================================================



    function deposit(uint256 amount) public {

        require(amount > 0, "Amount must be greater than zero");

        bool success = token.transferFrom(
            msg.sender,
            address(this),
            amount
        );

        require(success, "Transfer failed");

        balances[msg.sender] += amount;
    }


    // ==================================================
    // 2. NORMAL WITHDRAW — NO FEE
    // ==================================================

    function withdraw(uint256 amount) public onlyAdmin {

        require(amount > 0, "Amount must be greater than zero");

        // 100 + 100 = 200

        // bank = 200

        require(
            balances[msg.sender] >= amount,
            "Insufficient balance"
        );

        balances[msg.sender] -= amount;

        bool success = token.transfer(
            msg.sender,
            amount
        );

        require(success, "Transfer failed");
    }


    // ==================================================
    // 3. DEPOSIT WITH 2% FEE
    // ==================================================

    function depositWithFee(uint256 amount) public {

        require(amount > 0, "Amount must be greater than zero");

        // B = 100 + 100
        // B = 100 + 98 = 198

        bool success = token.transferFrom(
            msg.sender,
            address(this),
            amount
        );

        require(success, "Transfer failed");


        uint256 fee =
            (amount * FEE_BPS) / BPS;

        uint256 amountAfterFee =
            amount - fee;


        balances[msg.sender] += amountAfterFee;

        collectedFees += fee;
    }


    // ==================================================
    // 4. WITHDRAW WITH 2% FEE
    // ==================================================

    function withdrawWithFee(uint256 amount) public onlyAdmin {

        require(amount > 0, "Amount must be greater than zero");

        require(
            balances[msg.sender] >= amount,
            "Insufficient balance"
        );


        uint256 fee =
            (amount * FEE_BPS) / BPS;

        uint256 amountAfterFee =
            amount - fee;


        // Deduct full amount from user's vault balance
        balances[msg.sender] -= amount;

        // Record the fee
        collectedFees += fee;


        // Send remaining amount to user
        bool success = token.transfer(
            msg.sender,
            amountAfterFee
        );

        require(success, "Transfer failed");
    }


    // ==================================================
    // HELPER FUNCTIONS
    // ==================================================

    function getMyBalance()
        public
        view
        returns (uint256)
    {
        return balances[msg.sender];
    }


    function getVaultTokenBalance()
        public
        view
        returns (uint256)
    {
        return token.balanceOf(address(this));
    }


    function calculateFee(uint256 amount)
        public
        pure
        returns (
            uint256 fee,
            uint256 amountAfterFee
        )
    {
        fee = (amount * FEE_BPS) / BPS;

        amountAfterFee =
            amount - fee;
    }
}

import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";

contract SimpleStaking {

    using SafeERC20 for IERC20;

    IERC20 public stakingToken;

    // 1000 basis points = 10% APY
    uint256 public constant APY = 1000;
    uint256 public constant BPS = 10000;

    uint256 public constant LOCK_TIME = 5 minutes;
    uint256 public constant COOLDOWN_TIME = 30 seconds; // request for withdraw,

    address public owner;

    //  A => 100 , on 1st of OCtober, 

    struct Stake {
        uint256 amount;
        uint256 startTime;
        uint256 cooldownEnd;
        uint256 rewardAtRequest;
        bool withdrawalRequested; // true / false // A withdrawn 100,
    }

    mapping(address => Stake) public stakes;

    uint256 public rewardPool;


    constructor(address _token) {
        stakingToken = IERC20(_token);
        owner = msg.sender;
    }

    // A -> 100, 50 => 150 
    // B -> 100, 50 => 150 
    // TA = 200, => 200 - 300 = - 100 + 100 = 0; 


    // --------------------------------------------------
    // FUND REWARD POOL
    // --------------------------------------------------

    function fundRewards(uint256 amount) public {

        require(msg.sender == owner, "Only owner");

        stakingToken.safeTransferFrom(
            msg.sender,
            address(this),
            amount
        );

        rewardPool += amount;
    }


    // --------------------------------------------------
    // STAKE TOKENS
    // --------------------------------------------------

    function stake(uint256 amount) public {

        require(amount > 0, "Amount must be greater than zero");

        require(
            stakes[msg.sender].amount == 0,
            "Already staking"
        );

        stakingToken.safeTransferFrom(
            msg.sender,
            address(this),
            amount
        );

        stakes[msg.sender] = Stake({
            amount: amount,
            startTime: block.timestamp,
            cooldownEnd: 0,
            rewardAtRequest: 0,
            withdrawalRequested: false
        });
    }


    // --------------------------------------------------
    // CALCULATE CURRENT REWARD
    // --------------------------------------------------

    function calculateReward(address user)
        public
        view
        returns (uint256)
    {
        Stake memory userStake = stakes[user];

        if (userStake.amount == 0) {
            return 0;
        }

        // Stop earning when withdrawal is requested
        if (userStake.withdrawalRequested) {
            return userStake.rewardAtRequest;
        }

        uint256 timeStaked =
            block.timestamp - userStake.startTime;

        uint256 reward =
            ( userStake.amount * APY * timeStaked) / ( BPS * 365 days);

        return reward;
    }


    // --------------------------------------------------
    // REQUEST WITHDRAWAL
    // --------------------------------------------------

    function requestWithdrawal() public {

        Stake storage userStake = stakes[msg.sender];

        require(
            userStake.amount > 0,
            "No active stake"
        );

        require( block.timestamp >= userStake.startTime + LOCK_TIME, "Tokens are still locked");

        require(
            !userStake.withdrawalRequested,
            "Already requested"
        );

        // Freeze reward at this point
        uint256 reward =
            calculateReward(msg.sender);

        userStake.rewardAtRequest = reward;

        userStake.withdrawalRequested = true;

        userStake.cooldownEnd = block.timestamp + COOLDOWN_TIME;
    }

    // --------------------------------------------------
    // WITHDRAW PRINCIPAL + REWARD
    // --------------------------------------------------

    function withdraw() public {

        Stake memory userStake = stakes[msg.sender];

        require(
            userStake.withdrawalRequested,
            "Request withdrawal first"
        );

        require(
            block.timestamp >= userStake.cooldownEnd,
            "Cooldown is not finished"
        );

        require(
            rewardPool >= userStake.rewardAtRequest,
            "Not enough rewards"
        );

        uint256 principal = userStake.amount;
        uint256 reward = userStake.rewardAtRequest;

        rewardPool -= reward;

        // Delete before external transfer
        delete stakes[msg.sender];

        stakingToken.transfer(
            msg.sender,
            principal + reward
        );
    }


    // --------------------------------------------------
    // CHECK WHEN TOKENS UNLOCK
    // --------------------------------------------------

    function unlockTime(address user)
        public
        view
        returns (uint256)
    {
        return stakes[user].startTime + LOCK_TIME;
    }
}
