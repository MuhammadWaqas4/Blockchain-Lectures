pragma solidity ^0.8.0;

contract ModifierContract {
    uint public value;

    modifier onlyPositive(uint _val) {
        if (_val <= 0) revert("Must be positive");
        _;
    }

    function setValue(uint _val) public onlyPositive(_val) {
        value = _val;
    }
}

contract AccessControlContract {
    address public owner;

    constructor() {
        owner = msg.sender;
    }

    modifier onlyOwner() {
        require(msg.sender == owner, "Not owner");
        _;
    }

    function changeOwner(address _newOwner) public onlyOwner {
        owner = _newOwner;
    }
}

contract RequireContract {
    uint public balance = 100;

    function withdraw(uint _amount) public {
        require(_amount <= balance, "Insufficient balance");
        balance -= _amount;
    }
}

contract RevertContract {
    uint public balance = 100;

    function withdraw(uint _amount) public {
        if (_amount > balance) {
            revert("Insufficient balance");
        }
        balance -= _amount;
    }
}

contract AssertContract {
    uint public totalSupply = 100;

    function burn(uint _amount) public {
        require(totalSupply >= _amount, "Not enough");
        totalSupply -= _amount;
        assert(totalSupply >= 0);
    }

    function divide(uint a, uint b) public pure returns (uint) {
        require(b != 0, "Zero division");
        uint result = a / b;
        assert(result * b <= a);
        return result;
    }
}

contract BaseReusable {
    uint public baseValue;

    modifier onlyValid(uint _val) {
        require(_val > 0, "Invalid");
        _;
    }

    function setBase(uint _val) public onlyValid(_val) {
        baseValue = _val;
    }
}

contract CodeReuseContract is BaseReusable {
    function setValue(uint _val) public onlyValid(_val) {
        baseValue = _val * 2;
    }
}

contract BestPracticesContract {
    mapping(address => uint) public balances;

    function deposit() public payable {
        balances[msg.sender] += msg.value;
    }

    function withdraw(uint _amount) public {
        require(_amount <= balances[msg.sender], "Insufficient balance");
        balances[msg.sender] -= _amount;
        payable(msg.sender).transfer(_amount);
    }
}
